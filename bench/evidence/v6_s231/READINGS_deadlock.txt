from mw.bench import Caliber
import json
c = Caliber(memory_seed=(0,0), repeat=1)
disk = Caliber(**{k: v for k, v in json.loads(json.dumps(c.as_dict())).items()})
print('sig 相同 :', c.signature()==disk.signature())
print('same_as :', c.same_as(disk), '| diff:', c.diff(disk))
